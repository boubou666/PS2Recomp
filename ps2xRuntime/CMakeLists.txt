            NAMES "${library_name}" "${library_name}.a" "lib${library_name}.a"

function(ps2x_stage_ffmpeg_runtime_dlls target_name)
    if(NOT WIN32)
        return()
    endif()

    if(NOT PS2X_FFMPEG_BIN_DIR OR NOT PS2X_COPY_FFMPEG_DLLS_SCRIPT)
        return()
    endif()

    if(TARGET ffmpeg_external)
        add_dependencies(${target_name} ffmpeg_external)
    endif()

    add_custom_command(
        TARGET ${target_name} POST_BUILD
        COMMAND "${CMAKE_COMMAND}"
        "-Dsource_dir=${PS2X_FFMPEG_BIN_DIR}"
        "-Ddest_dir=$<TARGET_FILE_DIR:${target_name}>"
        -P "${PS2X_COPY_FFMPEG_DLLS_SCRIPT}"
        VERBATIM
    )
endfunction()

# The runtime (ps2_runtime_macros.h, GS/VU paths) uses SSE4.1/AVX2 intrinsics. MSVC gets
# /arch:AVX2 from EnableFastReleaseMode, but GCC/Clang need the target flag in every config
# or the always_inline intrinsics fail to compile. PUBLIC so the runner and generated code
# see the same ISA.
if(CMAKE_SYSTEM_PROCESSOR MATCHES "x86_64|AMD64|x86|i[3-6]86" AND NOT MSVC AND NOT PS2X_IS_ARM_TARGET)
    set(PS2X_X86_MARCH "x86-64-v3" CACHE STRING
        "-march value for the runtime and recompiled code on GCC/Clang (x86-64-v3 = AVX2/FMA/BMI)")
    target_compile_options(ps2_runtime PUBLIC "-march=${PS2X_X86_MARCH}")
endif()

# Opt-in sampling profiler (PS2X_PROFILE=<seconds>, Windows and Linux). It includes <windows.h>, which
# clashes with raylib names inside unity batches, so it is compiled on its own.
if((WIN32 OR CMAKE_SYSTEM_NAME STREQUAL "Linux") AND NOT PS2X_IS_ANDROID AND NOT PS2X_IS_VITA)
    set(PS2X_SAMPLING_PROFILER_SOURCE "${CMAKE_CURRENT_SOURCE_DIR}/src/profiling/sampling_profiler.cpp")
    target_sources(ps2EntryRunner PRIVATE "${PS2X_SAMPLING_PROFILER_SOURCE}")
    set_source_files_properties("${PS2X_SAMPLING_PROFILER_SOURCE}" PROPERTIES SKIP_UNITY_BUILD_INCLUSION ON)
    if(WIN32)
        # raylib first: winmm exports a PlaySound too, and ahead of raylib on a MinGW link line it takes raylib's
        # PlaySound calls, then clashes with raudio's definition.
        target_link_libraries(ps2EntryRunner PRIVATE raylib winmm)
    endif()
endif()

# "Fast" is a local /O2 config without LTCG (the generated code is too large to link with /GL in
# reasonable time); give the runtime library itself full inlining, it holds the software rasterizer.
if(MSVC AND CMAKE_BUILD_TYPE STREQUAL "Fast")
    target_compile_options(ps2_runtime PRIVATE /Ob2 /Oi /GS-)
endif()

if(CMAKE_BUILD_TYPE STREQUAL "Release" OR CMAKE_BUILD_TYPE STREQUAL "RelWithDebInfo")
    EnableFastReleaseMode(ps2_runtime)
    # The runtime library holds the interpreters (VU0/VU1, IOP) and the software rasterizer: -O3 for it even in
    # RelWithDebInfo (-O2). The generated code keeps the configuration's level (it is too large for -O3 builds).
    if(CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
        target_compile_options(ps2_runtime PRIVATE -O3)
    endif()
    EnableFastReleaseMode(ps2EntryRunner)

    if(WIN32 AND NOT PS2X_SHOW_WINDOWS_CONSOLE)
        if(MSVC)
            target_link_options(ps2EntryRunner PRIVATE
                $<$<OR:$<CONFIG:Release>,$<CONFIG:RelWithDebInfo>>:/SUBSYSTEM:WINDOWS>
                $<$<OR:$<CONFIG:Release>,$<CONFIG:RelWithDebInfo>>:/ENTRY:mainCRTStartup>
            )
        elseif(MINGW)
            target_link_options(ps2EntryRunner PRIVATE
                $<$<OR:$<CONFIG:Release>,$<CONFIG:RelWithDebInfo>>:-mwindows>
            )
        endif()
    endif()
endif()
